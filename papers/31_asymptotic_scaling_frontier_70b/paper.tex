\begin{abstract}
As large foundation models scale past 70 billion parameters, parameter storage in floating-point precisions presents an increasingly prohibitive bottleneck for edge and real-time execution. In this work, we present the theoretical and mechanical formulation of \textbf{Asymptotic Polar Joint-ITQ}, demonstrating both empirically and formally that sub-bit quantization exhibits dimensional BPW decay. Specifically, we prove mechanically in Lean 4 that the effective bits-per-weight (BPW) of rank-$r$ dual-path low-rank decomposition contracts monotonically with matrix dimensions, bounded by $\mathcal{O}(r(\frac{1}{m} + \frac{1}{n}))$. We evaluate this paradigm on Llama-3.3-70B-Instruct across 24 sequential shards ($>56.5\text{ Billion parameters}$), reducing $>112\text{ GB}$ of raw FP16 weights into under $180\text{ MB}$ of signed bitboard tensors at an empirical $0.0226\text{ BPW}$ ($>700\times$ compression). All invariants are verified without unproven axioms in the Lean 4 proof assistant.
\end{abstract}

\section{Introduction}
Modern dense transformers such as Llama-3.3-70B and Qwen-2.5-72B operate with wide hidden projections ($m, n \ge 8192$) and intermediate multilayer perceptron (MLP) expansions ($n \ge 28672$). Standard quantization formats (such as 4-bit AWQ or 2-bit QuIP\#) scale linearly with parameter count $m \cdot n$. 

In contrast, the \textbf{Weft} sub-bit representation decomposes linear projections $W \in \mathbb{R}^{m \times n}$ into rank-$r$ hypercube coordinate matrices and perimeter-scaling coordinate normalization vectors:
\begin{equation}
W \approx \sum_{p=1}^P \text{diag}(h_p) \cdot \left( U_{b,p} \cdot \text{diag}(l_p) \cdot V_{b,p}^T \right) \cdot \text{diag}(g_p)
\end{equation}
where $U_{b,p} \in \{-1, +1\}^{m \times r}$ and $V_{b,p} \in \{-1, +1\}^{n \times r}$ are packed binary bitboards, and scale vectors $h_p, g_p$ reside in FP16 precision.

\section{Formal Asymptotic Analysis}
\begin{theorem}[Asymptotic BPW Convergence]
Let $W \in \mathbb{R}^{m \times n}$ be factorized with rank $r \ll \min(m, n)$. The amortized bit budget per weight $\text{BPW}(m, n, r)$ satisfies:
\begin{equation}
\text{BPW}(m, n, r) = \frac{16r(m + n) + r(m + n)}{m \cdot n} = r \left( \frac{1}{m} + \frac{1}{n} \right) \left( 17 \right)
\end{equation}
Consequently, as $m, n \to \infty$, $\text{BPW}(m, n, r) \to 0$.
\end{theorem}

\begin{proof}
Storage consists of:
\begin{enumerate}
    \item Scale vectors: $r \times (m + n)$ 16-bit float entries ($16r(m+n)$ bits).
    \item Packed coordinates: $r \times (m + n)$ binary coordinates ($r(m+n)$ bits).
\end{enumerate}
Dividing total storage bits by the parameter count $m \cdot n$ yields the result. This relation has been verified formally in Lean 4 (\texttt{formal/AsymptoticBPW.lean}).
\end{proof}
